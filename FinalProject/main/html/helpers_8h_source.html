106void IRAM_ATTR buzzTimerISR() {
107 if (!buzzEnabled) {
108 ledcWrite(buzzerPin, 0);
109 return;
110 }
111
112 buzzTickCount++;
113 if (buzzTickCount > BUZZ_DURATION_TICKS) {
114 buzzEnabled = false;
115 buzzTickCount = 0;
116 ledcWrite(buzzerPin, 0);
117 return;
118 }
119
120 static bool buzzState = false;
121 buzzState = !buzzState;
122 ledcWrite(buzzerPin, buzzState ? BUZZ_DUTY_ON : 0);
123}

132void IRAM_ATTR estopISR() {
133 unsigned long now = millis();
134 if (now - lastDebounceTime < DEBOUNCE_MS) return;
135 lastDebounceTime = now;
136 estopActive = (digitalRead(estopPin) == LOW);
137}

174bool stepDegrees(float degrees) {
175 int steps = (int)(degrees * STEPS_PER_DEG);
176 int dir = (steps > 0) ? 1 : -1;
177 steps = abs(steps);
178
179 for (int s = 0; s < steps; s++) {
180 xSemaphoreTake(stateMutex, portMAX_DELAY);
181 float angle = stepperAngle;
182 xSemaphoreGive(stateMutex);
183
184 float newAngle = angle + dir * (360.0 / 2048.0);
185 if (newAngle > STEPPER_MAX_DEG || newAngle < -STEPPER_MAX_DEG) {
186 releaseStepper();
187 return false;
188 }
189
190 myStepper.step(dir);
191
192 xSemaphoreTake(stateMutex, portMAX_DELAY);
193 stepperAngle = newAngle;
194 xSemaphoreGive(stateMutex);
195
196 vTaskDelay(pdMS_TO_TICKS(STEP_DELAY_MS));
197
198 // Abort mid-move if state changes
199 if (systemState == LOCKED || systemState == ESTOP) {
200 releaseStepper();
201 return false;
202 }
203 }
204
205 releaseStepper();
206 return true;
207}

215void stepperToCenter() {
216 xSemaphoreTake(stateMutex, portMAX_DELAY);
217 float angle = stepperAngle;
218 xSemaphoreGive(stateMutex);
219
220 Serial.print("[MOTOR] Returning to center from ");
221 Serial.print(angle, 1);
222 Serial.println(" deg");
223
224 stepDegrees(-angle);
225
226 xSemaphoreTake(stateMutex, portMAX_DELAY);
227 stepperAngle = 0.0;
228 xSemaphoreGive(stateMutex);
229
230 releaseStepper();
231}

241float getDistance() {
242 digitalWrite(trigPin, LOW);
243 delayMicroseconds(2);
244 digitalWrite(trigPin, HIGH);
245 delayMicroseconds(10);
246 digitalWrite(trigPin, LOW);
247 long duration = pulseIn(echoPin, HIGH, 30000);
248 if (duration == 0) return -1.0;
249 return (duration / 2.0) * 0.0343;
250}

259float getStableDistance() {
260 float total = 0;
261 int valid = 0;
262 for (int i = 0; i < 3; i++) {
263 float d = getDistance();
264 if (d > 0) { total += d; valid++; }
265 vTaskDelay(pdMS_TO_TICKS(10));
266 }
267 return (valid > 0) ? total / valid : -1.0;
268}
